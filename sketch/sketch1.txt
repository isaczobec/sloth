    return c;
}

// A class can be an algebraic datatype. 
class ADTExample :=

    int b;

    // some operations; assumes a is First

};

g(Second a) := {
    // some operations; assumes a is First
    return a.e;
};

// -----------------------------------------
// ---- Example implementation of a BST ----
// -----------------------------------------

<T> interface Comparable := {
    bool greaterThan(T other);
}

<T> interface Equatable := {
    bool greaterThan(T other);
}

<T : Comparable<T>, Equatable<T>> class BinaryTree := 
Node {
    BinaryTree left = (Nothing) {}; // classes can be recursive
    BinaryTree right = (Nothing) {};
    T value;
} |
Nothing {

}

// everyything is passed by value unless & succeeds a type
// a function not marked impure cannot take by reference

// defining a variable to be the same as some 
// other variable will initially cause the second
// variable to point at the first variable in memory.
// making mutations to the variable, if that variable
// is not marked as a reference will create a copy.
// thus, since next is not touched before it is bound
// to a reference in the next recursive insert call
// is fine. 

// Perhaps there should be an operator forcing a variable
// to be copied/passed by value?

impure (BinaryTree&, T) -> BinaryTree& insert;
insert(Node n, t) := {

    BinaryTree& next := 
        t.greaterThan(n.value) ? n.right |
        true ? n.left;

    // writing a subtype of an ADT and then a variable name both checks a truth value
    // and binds that the variable is of that subtype in the opened scope
    if (Node next) { 
        return insert(next, t);
    } else if (Nothing next) {
        next = (BinaryTree Node) { value = t; };
    }
}
